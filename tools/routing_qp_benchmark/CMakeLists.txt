cmake_minimum_required(VERSION 3.24)
project(RoutingQpComparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT WIN32)
    message(FATAL_ERROR "This object-reuse experiment currently targets the qualified Windows build.")
endif()
file(READ "${AF_ROOT}/native/CMakeLists.txt" NATIVE_TARGETS)
string(REGEX MATCH "add_library\\(af_native SHARED[^)]*\\)" NATIVE_TARGET "${NATIVE_TARGETS}")
string(REGEX MATCHALL "src/[A-Za-z_]+\\.cpp" NATIVE_SOURCES "${NATIVE_TARGET}")
foreach(SOURCE IN LISTS NATIVE_SOURCES)
    if(NOT SOURCE STREQUAL "src/routing_solve.cpp")
        list(APPEND OBJECTS "${AF_ROOT}/build/dev/native/CMakeFiles/af_native.dir/${SOURCE}.obj")
    endif()
endforeach()
if(NOT OBJECTS)
    message(FATAL_ERROR "Build the normal native target first.")
endif()
add_library(af_routing_lab SHARED ${OBJECTS} "${AF_STAGE}/native/src/routing_solve.cpp")
set_source_files_properties(${OBJECTS} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
target_include_directories(af_routing_lab PRIVATE "${AF_ROOT}/native/src" "${AF_ROOT}/native/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(af_routing_lab SYSTEM PRIVATE "${AF_ROOT}/native/vendor")
target_compile_definitions(af_routing_lab PRIVATE AF_NATIVE_BUILD NOMINMAX _USE_MATH_DEFINES EIGEN_MPL2_ONLY EIGEN_DONT_PARALLELIZE)
target_compile_options(af_routing_lab PRIVATE /W4 /WX /fp:precise)
target_link_libraries(af_routing_lab PRIVATE "${AF_ROOT}/build/dev/native/af_curve_fit.lib" "${AF_ROOT}/build/dev/native/boolean/af_boolean.lib")
