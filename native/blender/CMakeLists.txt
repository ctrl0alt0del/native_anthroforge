set(BM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/blender_bmesh")
file(READ "${BM_ROOT}/sources.json" BM_MANIFEST)
string(JSON BM_COUNT LENGTH "${BM_MANIFEST}" sources)
math(EXPR BM_LAST "${BM_COUNT} - 1")
set(BM_SOURCES)
foreach(INDEX RANGE ${BM_LAST})
    string(JSON BM_FILE GET "${BM_MANIFEST}" sources ${INDEX})
    list(APPEND BM_SOURCES "${BM_ROOT}/${BM_FILE}")
endforeach()
add_library(af_bmesh STATIC ${BM_SOURCES})
set_target_properties(af_bmesh PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(af_bmesh PRIVATE cxx_std_17)
foreach(PART source/blender/bmesh source/blender/bmesh/intern source/blender/bmesh/tools
    source/blender/blenlib source/blender/blenlib/intern source/blender/blenkernel
    source/blender/makesdna intern/guardedalloc intern/guardedalloc/intern intern/atomic intern/clog)
    target_include_directories(af_bmesh SYSTEM PUBLIC "${BM_ROOT}/${PART}")
endforeach()
target_compile_definitions(af_bmesh PRIVATE NDEBUG NOMINMAX MATH_STANDALONE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(af_bmesh PRIVATE /W0 /fp:precise)
    target_include_directories(af_bmesh SYSTEM PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/compat")
else()
    target_compile_options(af_bmesh PRIVATE -w -ffp-contract=off)
endif()
