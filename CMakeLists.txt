cmake_minimum_required(VERSION 3.24)
project(AnthroForge VERSION 0.3.1 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(AF_BUILD_QT_BRIDGE "Build the Qt Quick 3D geometry adapter" ON)
set(AF_ADDON_SOURCE "" CACHE PATH "Optional existing addon checkout, read-only reference")

include(CTest)
add_subdirectory(native)
if(AF_BUILD_QT_BRIDGE)
    find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Quick Quick3D Qml ShaderTools QmlDebugPrivate)
    add_subdirectory(native/qt)
    add_subdirectory(native/profiling)
endif()
if(AF_ADDON_SOURCE)
    add_subdirectory("${AF_ADDON_SOURCE}/native/smoothing" "addon-smoothing")
endif()
